  <script>
  (function(){
    var nav=document.querySelector('[data-nav]');if(!nav)return;
    var btn=nav.querySelector('.burger');
    var desk=nav.querySelectorAll('[data-desktop-only]');
    if(window.matchMedia('(min-width:820px)').matches)desk.forEach(function(d){d.style.display='inline-flex'});
    btn.addEventListener('click',function(){
      var open=nav.getAttribute('aria-expanded')==='true';
      nav.setAttribute('aria-expanded',String(!open));
      btn.setAttribute('aria-expanded',String(!open));
    });
  })();</script>
